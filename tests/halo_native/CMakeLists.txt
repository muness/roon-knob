cmake_minimum_required(VERSION 3.16)
project(halo_native_checks C CXX)
set(CMAKE_C_STANDARD 11)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h" CACHE STRING "" FORCE)
if(NOT LVGL_SOURCE)
  message(FATAL_ERROR "Pass LVGL_SOURCE (HALO's pinned LVGL checkout)")
endif()
# SDK substitutes exist only in the native build tree. Keeping templates out of
# the production header index prevents them from impersonating ESP-IDF headers.
file(GLOB_RECURSE HALO_TEST_STUBS CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/stubs/*.h.in")
foreach(stub IN LISTS HALO_TEST_STUBS)
  file(RELATIVE_PATH stub_relative "${CMAKE_CURRENT_SOURCE_DIR}/stubs" "${stub}")
  string(REGEX REPLACE "\\.in$" "" stub_header "${stub_relative}")
  configure_file("${stub}" "${CMAKE_CURRENT_BINARY_DIR}/stubs/${stub_header}" COPYONLY)
endforeach()
# Exercise the production legacy-header removal with real LVGL, rather than
# a stub that would miss the overlapping target reported on hardware.
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../../common/ui.c" UI_SOURCE)
string(REGEX MATCH "void ui_remove_legacy_zone_touch_target\\(void\\) \{[^}]*\}" HEADER_TOUCH_REMOVAL "${UI_SOURCE}")
if(NOT HEADER_TOUCH_REMOVAL)
  message(FATAL_ERROR "Production legacy header removal function not found")
endif()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/stubs/header_touch_removal.h" "${HEADER_TOUCH_REMOVAL}\n")
add_subdirectory(${LVGL_SOURCE} lvgl)
add_executable(halo_native_checks test_halo_native.c ../../common/controller_connection.c ../../common/controller_library.c ../../common/controller_navigation.c ../../idf_app/main/fonts/lato_22.c ../../idf_app/main/fonts/notosans_28.c ../../idf_app/main/fonts/material_icons_22.c ../../idf_app/main/fonts/material_icons_28.c)
target_include_directories(halo_native_checks PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/stubs" ../../common ../../idf_app/main ${CJSON_SOURCE})
if(CJSON_SOURCE)
  target_sources(halo_native_checks PRIVATE ${CJSON_SOURCE}/cJSON.c)
else()
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(CJSON REQUIRED IMPORTED_TARGET libcjson)
  target_link_libraries(halo_native_checks PRIVATE PkgConfig::CJSON)
endif()
target_link_libraries(halo_native_checks PRIVATE lvgl m)
option(HALO_SANITIZE "Instrument production renderer and LVGL for address/undefined behavior checks" OFF)
if(HALO_SANITIZE)
  target_compile_options(lvgl PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_compile_options(halo_native_checks PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(halo_native_checks PRIVATE -fsanitize=address,undefined)
endif()

# Same production renderer with the Dial capability adapter enabled.
add_executable(dial_native_checks test_halo_native.c ../../common/controller_connection.c ../../common/controller_library.c ../../common/controller_navigation.c ../../idf_app/main/fonts/lato_22.c ../../idf_app/main/fonts/notosans_28.c ../../idf_app/main/fonts/material_icons_22.c ../../idf_app/main/fonts/material_icons_28.c)
target_compile_definitions(dial_native_checks PRIVATE TEST_DIAL_TOUCH=1)
target_include_directories(dial_native_checks PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/stubs" ../../common ../../idf_app/main ${CJSON_SOURCE})
if(CJSON_SOURCE)
  target_sources(dial_native_checks PRIVATE ${CJSON_SOURCE}/cJSON.c)
else()
  target_link_libraries(dial_native_checks PRIVATE PkgConfig::CJSON)
endif()
target_link_libraries(dial_native_checks PRIVATE lvgl m)
if(HALO_SANITIZE)
  target_compile_options(dial_native_checks PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(dial_native_checks PRIVATE -fsanitize=address,undefined)
endif()
